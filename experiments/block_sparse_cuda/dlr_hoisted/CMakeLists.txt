# Isolated experiment; the production CPU and CUDA evaluators are unchanged.
xca_add_benchmark(block_sparse_dlr_hoisted)

# Generate a private control from current production sources. Never edit or
# shadow installed headers; fail if the intended two-site adaptation changes.
set(control_header "${PROJECT_SOURCE_DIR}/c++/triqs_xca/block_sparse/diagram_evaluator.hpp")
set(control_source "${PROJECT_SOURCE_DIR}/c++/triqs_xca/block_sparse/diagram_evaluator.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${control_header}" "${control_source}")
file(READ "${control_header}" header_text)
file(READ "${control_source}" source_text)
string(REGEX MATCHALL "itops\\.convolve\\(" sites "${source_text}")
list(LENGTH sites site_count)
if(NOT site_count EQUAL 2)
  message(FATAL_ERROR "Hoisted evaluator control expects exactly two convolution sites")
endif()
string(FIND "${header_text}" "imtime_ops itops;" member_position)
if(member_position EQUAL -1)
  message(FATAL_ERROR "Hoisted evaluator control cannot identify rank-operations member")
endif()
string(REPLACE "DiagramEvaluator" "HoistedDiagramEvaluator" header_text "${header_text}")
string(REPLACE "DiagramEvaluator" "HoistedDiagramEvaluator" source_text "${source_text}")
string(REPLACE "imtime_ops itops;" "block_sparse_dlr_experiment::ImtimeControl itops;" header_text "${header_text}")
string(REPLACE "#pragma once" "#pragma once\n#include \"hoisted.hpp\"" header_text "${header_text}")
string(REPLACE "\"triqs_xca/block_sparse/diagram_evaluator.hpp\"" "\"hoisted_evaluator.hpp\"" source_text "${source_text}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/hoisted_evaluator.hpp" CONTENT "${header_text}" @ONLY)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/hoisted_evaluator.cpp" CONTENT "${source_text}" @ONLY)
xca_add_benchmark(block_sparse_cpu_hoisted_benchmark)
target_sources(block_sparse_cpu_hoisted_benchmark PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/hoisted_evaluator.cpp")
target_include_directories(block_sparse_cpu_hoisted_benchmark PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
