if(NOT nda_WITH_CUDA_SUPPORT)
  return()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(cuda_dir "${PROJECT_SOURCE_DIR}/c++/triqs_xca/block_sparse/cuda")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${cuda_dir}/diagram_evaluator.hpp" "${cuda_dir}/diagram_evaluator.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${PROJECT_SOURCE_DIR}/experiments/block_sparse_cuda/phase_liveness/generate.py" "${PROJECT_SOURCE_DIR}/experiments/block_sparse_cuda/host_inputs/generate.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py"
  "${PROJECT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
add_library(block_sparse_stream_fragment_evaluator STATIC "${CMAKE_CURRENT_BINARY_DIR}/stream_evaluator.cpp")
target_include_directories(block_sparse_stream_fragment_evaluator PUBLIC "${CMAKE_CURRENT_BINARY_DIR}" "${cuda_dir}")
target_link_libraries(block_sparse_stream_fragment_evaluator PUBLIC ${PROJECT_NAME}::${PROJECT_NAME}_c)
add_executable(block_sparse_stream_fragment_benchmark benchmark.cpp)
target_link_libraries(block_sparse_stream_fragment_benchmark PRIVATE block_sparse_stream_fragment_evaluator)
add_executable(block_sparse_stream_fragment_cutoff_audit cutoff_audit.cpp)
target_link_libraries(block_sparse_stream_fragment_cutoff_audit PRIVATE block_sparse_stream_fragment_evaluator)
if(Build_Tests)
  set(reference "${PROJECT_SOURCE_DIR}/test/c++/block_sparse/cuda_diagram_evaluator.cpp")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${reference}")
  file(READ "${reference}" tests)
  string(REPLACE "<triqs_xca/block_sparse/cuda/diagram_evaluator.hpp>" "\"stream_evaluator.hpp\"" tests "${tests}")
  string(REPLACE "gpu::DiagramEvaluator" "gpu::StreamFragmentEvaluator" tests "${tests}")
  file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/inherited_tests.cpp" CONTENT "${tests}" @ONLY)
  add_executable(block_sparse_stream_fragment_tests "${CMAKE_CURRENT_BINARY_DIR}/inherited_tests.cpp" tests.cpp)
  target_link_libraries(block_sparse_stream_fragment_tests PRIVATE block_sparse_stream_fragment_evaluator ${PROJECT_NAME}_test_utils gtest_main)
  add_test(NAME "experiments/block_sparse_cuda/stream_fragments" COMMAND block_sparse_stream_fragment_tests)
endif()
