if(NOT nda_WITH_CUDA_SUPPORT)
  return()
endif()
set(source_dir "${PROJECT_SOURCE_DIR}/c++/triqs_xca/block_sparse/cuda")
set(reference "${PROJECT_SOURCE_DIR}/test/c++/block_sparse/cuda_diagram_evaluator.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/experiments/block_sparse_cuda/reference_snapshot/diagram_evaluator.hpp" "${PROJECT_SOURCE_DIR}/experiments/block_sparse_cuda/reference_snapshot/diagram_evaluator.cpp" "${reference}" "${CMAKE_CURRENT_SOURCE_DIR}/special.inc")
file(READ "${PROJECT_SOURCE_DIR}/experiments/block_sparse_cuda/reference_snapshot/diagram_evaluator.hpp" header)
file(READ "${PROJECT_SOURCE_DIR}/experiments/block_sparse_cuda/reference_snapshot/diagram_evaluator.cpp" source)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/special.inc" special)
string(FIND "${source}" "    void special(" first)
string(FIND "${source}" "    void evaluate(" last)
if(first LESS 0 OR last LESS first)
  message(FATAL_ERROR "Kappa reuse cannot identify the reference special vertex")
endif()
string(SUBSTRING "${source}" 0 ${first} prefix)
string(SUBSTRING "${source}" ${last} -1 suffix)
set(source "${prefix}${special}${suffix}")
string(FIND "${source}" "kernel_bytes, mapped_storage_bound());" bound_site)
if(bound_site LESS 0)
  message(FATAL_ERROR "Kappa reuse cannot identify scratch admission")
endif()
string(REPLACE "kernel_bytes, mapped_storage_bound());" "kernel_bytes, mapped_storage_bound(), checked_add(7, kappa_slots()));" source "${source}")
string(REPLACE "DiagramEvaluator" "KappaCacheEvaluator" header "${header}")
string(REPLACE "DiagramEvaluator" "KappaCacheEvaluator" source "${source}")
string(REPLACE "\"diagram_evaluator.hpp\"" "\"kappa_evaluator.hpp\"" source "${source}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/kappa_evaluator.hpp" CONTENT "${header}" @ONLY)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/kappa_evaluator.cpp" CONTENT "${source}" @ONLY)
if(Build_Tests)
  file(READ "${reference}" tests)
  string(REPLACE "<triqs_xca/block_sparse/cuda/diagram_evaluator.hpp>" "\"kappa_evaluator.hpp\"" tests "${tests}")
  string(REPLACE "gpu::DiagramEvaluator" "gpu::KappaCacheEvaluator" tests "${tests}")
  file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/kappa_tests.cpp" CONTENT "${tests}" @ONLY)
endif()
foreach(slots IN ITEMS 2 16)
  add_library(block_sparse_kappa_${slots} STATIC "${CMAKE_CURRENT_BINARY_DIR}/kappa_evaluator.cpp")
  target_include_directories(block_sparse_kappa_${slots} PUBLIC "${CMAKE_CURRENT_BINARY_DIR}" "${source_dir}")
  target_compile_definitions(block_sparse_kappa_${slots} PRIVATE XCA_EXPERIMENT_KAPPA_SLOTS=${slots})
  target_link_libraries(block_sparse_kappa_${slots} PUBLIC ${PROJECT_NAME}::${PROJECT_NAME}_c)
  add_executable(block_sparse_kappa_${slots}_benchmark benchmark.cpp)
  target_compile_definitions(block_sparse_kappa_${slots}_benchmark PRIVATE XCA_EXPERIMENT_KAPPA_SLOTS=${slots})
  target_link_libraries(block_sparse_kappa_${slots}_benchmark PRIVATE block_sparse_kappa_${slots} ${PROJECT_NAME}_warnings)
  if(Build_Tests)
    add_executable(block_sparse_kappa_${slots}_tests "${CMAKE_CURRENT_BINARY_DIR}/kappa_tests.cpp" qualification.cpp)
    target_compile_definitions(block_sparse_kappa_${slots}_tests PRIVATE XCA_EXPERIMENT_KAPPA_SLOTS=${slots})
    target_link_libraries(block_sparse_kappa_${slots}_tests PRIVATE block_sparse_kappa_${slots} ${PROJECT_NAME}_test_utils gtest_main)
    add_test(NAME "experiments/block_sparse_cuda/kappa_${slots}" COMMAND block_sparse_kappa_${slots}_tests)
  endif()
endforeach()
