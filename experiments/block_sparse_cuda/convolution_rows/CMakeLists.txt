if(NOT nda_WITH_CUDA_SUPPORT)
  return()
endif()
set(cuda_dir "${PROJECT_SOURCE_DIR}/c++/triqs_xca/block_sparse/cuda")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${cuda_dir}/dlr_ops.hpp" "${cuda_dir}/dlr_ops.cpp")
file(READ "${cuda_dir}/dlr_ops.hpp" header)
file(READ "${cuda_dir}/dlr_ops.cpp" source)
string(REPLACE "DlrOps" "RowDlrOps" header "${header}")
string(REPLACE "DlrOps" "RowDlrOps" source "${source}")
string(REPLACE "    // Coefficient inputs, time-ordered value output" "    // Row experiment requires disjoint output; bound excludes caller tensors.\n    std::size_t row_storage_bound(long M, long K, long N, long rows) const;\n    void convolve_rows(double beta, TensorConstView fc, TensorConstView gc, TensorView out, long rows);\n    // Coefficient inputs, time-ordered value output" header "${header}")
string(REPLACE "\"dlr_ops.hpp\"" "\"row_dlr.hpp\"" source "${source}")
set(source "${source}\n#include \"rows.inc\"\n")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/row_dlr.hpp" CONTENT "${header}" @ONLY)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/row_dlr.cpp" CONTENT "${source}" @ONLY)
add_library(block_sparse_row_dlr STATIC "${CMAKE_CURRENT_BINARY_DIR}/row_dlr.cpp")
target_include_directories(block_sparse_row_dlr PUBLIC "${CMAKE_CURRENT_BINARY_DIR}" "${cuda_dir}" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(block_sparse_row_dlr PUBLIC ${PROJECT_NAME}::${PROJECT_NAME}_c)
add_executable(block_sparse_row_dlr_benchmark benchmark.cpp)
target_link_libraries(block_sparse_row_dlr_benchmark PRIVATE block_sparse_row_dlr)
if(Build_Tests)
  add_executable(block_sparse_row_dlr_tests tests.cpp)
  target_link_libraries(block_sparse_row_dlr_tests PRIVATE block_sparse_row_dlr gtest_main)
  add_test(NAME "experiments/block_sparse_cuda/convolution_rows" COMMAND block_sparse_row_dlr_tests)
endif()
