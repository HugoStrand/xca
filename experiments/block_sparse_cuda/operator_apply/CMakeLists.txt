# The experiment has native kernels but never changes production dispatch.
if(NOT nda_WITH_CUDA_SUPPORT OR NOT CMAKE_CUDA_COMPILER)
  return()
endif()
enable_language(CUDA)
find_package(CUDAToolkit REQUIRED)
add_library(block_sparse_operator_kernels STATIC kernels.cu)
set_target_properties(block_sparse_operator_kernels PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
target_compile_options(block_sparse_operator_kernels PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
target_link_libraries(block_sparse_operator_kernels PUBLIC CUDA::cudart)
xca_add_benchmark(block_sparse_operator_apply)
target_link_libraries(block_sparse_operator_apply PRIVATE block_sparse_operator_kernels)

# Private complete evaluator: retain all reference code except snapshot routing.
set(cuda_source_dir "${PROJECT_SOURCE_DIR}/c++/triqs_xca/block_sparse/cuda")
set(header_path "${cuda_source_dir}/diagram_evaluator.hpp")
set(source_path "${cuda_source_dir}/diagram_evaluator.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${header_path}" "${source_path}")
file(READ "${header_path}" header_text)
file(READ "${source_path}" source_text)
string(REGEX MATCHALL "DeviceSparseBlock" sites "${source_text}")
list(LENGTH sites site_count)
if(NOT site_count EQUAL 3)
  message(FATAL_ERROR "Ordered evaluator expects three sparse snapshot references")
endif()
string(REPLACE "DiagramEvaluator" "OrderedDiagramEvaluator" header_text "${header_text}")
string(REPLACE "DiagramEvaluator" "OrderedDiagramEvaluator" source_text "${source_text}")
string(REPLACE "DeviceSparseBlock" "block_sparse_operator_experiment::OrderedSparseBlock" source_text "${source_text}")
string(REPLACE "\"diagram_evaluator.hpp\"" "\"ordered_evaluator.hpp\"\n#include \"ordered_block.hpp\"" source_text "${source_text}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/ordered_evaluator.hpp" CONTENT "${header_text}" @ONLY)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/ordered_evaluator.cpp" CONTENT "${source_text}" @ONLY)
add_library(block_sparse_ordered_evaluator STATIC "${CMAKE_CURRENT_BINARY_DIR}/ordered_evaluator.cpp")
target_include_directories(block_sparse_ordered_evaluator PUBLIC "${CMAKE_CURRENT_BINARY_DIR}"
  PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${cuda_source_dir}")
target_link_libraries(block_sparse_ordered_evaluator PUBLIC ${PROJECT_NAME}::${PROJECT_NAME}_c block_sparse_operator_kernels)
xca_add_benchmark(block_sparse_ordered_evaluator_benchmark)
target_include_directories(block_sparse_ordered_evaluator_benchmark PRIVATE "${cuda_source_dir}")
target_link_libraries(block_sparse_ordered_evaluator_benchmark PRIVATE block_sparse_ordered_evaluator)
