cmake_minimum_required(VERSION 3.22)
project(xca_power_two LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
set(XCA_POW2_DOT_TILE "0" CACHE STRING "Grouped real-dot tile, 0 for whole dimension")
if(NOT XCA_POW2_DOT_TILE MATCHES "^(0|2|4|8|16)$")
 message(FATAL_ERROR "XCA_POW2_DOT_TILE must be 0, 2, 4, 8 or 16")
endif()
add_library(fused_kernels STATIC kernels.cu)
target_compile_definitions(fused_kernels PUBLIC XCA_POW2_DOT_TILE=${XCA_POW2_DOT_TILE})
set_target_properties(fused_kernels PROPERTIES CUDA_STANDARD 17)
target_compile_options(fused_kernels PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=true> $<$<COMPILE_LANGUAGE:CUDA>:-Xptxas=-v>)
target_link_libraries(fused_kernels PUBLIC CUDA::cudart)
find_package(TRIQS 4.0 REQUIRED)
find_package(cppdlr REQUIRED)
get_filename_component(XCA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(XCA_BASELINE_BUILD "${XCA_ROOT}/cbuild-cuda" CACHE PATH "Frozen CUDA build")
add_library(power_two STATIC dlr_ops.cpp diagram_evaluator.cpp device_ops.cpp)
target_include_directories(power_two PUBLIC "${XCA_ROOT}/c++")
target_compile_features(power_two PUBLIC cxx_std_23)
target_link_libraries(power_two PUBLIC triqs cppdlr::cppdlr_c fused_kernels)
add_executable(power_two_benchmark benchmark.cpp)
target_link_libraries(power_two_benchmark PRIVATE power_two "${XCA_BASELINE_BUILD}/c++/triqs_xca/libtriqs_xca_c.a"
 "${XCA_BASELINE_BUILD}/c++/triqs_soehyb/libtriqs_soehyb_c.a")
# Reuse existing analytic/rotation tests; the adapter replaces only ordinary SE.
enable_testing()
set(TEST_ROOT "${XCA_ROOT}/test/c++")
foreach(name one_fermion_se_spgf_all_evals two_fermions_se_spgf_all_evals unitary_invariance_all_evals)
 file(READ "${TEST_ROOT}/${name}.cpp" contents)
 string(REPLACE "dense::cuda::DiagramEvaluator" "FusedTestAdapter" contents "${contents}")
 file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp" "#include \"test_adapter.hpp\"\n${contents}")
 add_executable(batch_${name} "${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp")
 target_include_directories(batch_${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${TEST_ROOT}"
   "${XCA_BASELINE_BUILD}/deps/GTest_src/googletest/include")
 target_link_libraries(batch_${name} PRIVATE power_two
   "${XCA_BASELINE_BUILD}/lib/libgtest_main.a"
   "${XCA_BASELINE_BUILD}/test/c++/libtriqs_xca_test_utils.a"
   "${XCA_BASELINE_BUILD}/lib/libgtest.a"
   "${XCA_BASELINE_BUILD}/c++/triqs_xca/libtriqs_xca_c.a")
 add_test(NAME batch_${name} COMMAND batch_${name})
endforeach()
add_executable(batch_dlr test_dlr.cpp ../serial_stream/device_ops.cpp ../serial_stream/dlr_ops.cpp)
target_include_directories(batch_dlr PRIVATE "${XCA_BASELINE_BUILD}/deps/GTest_src/googletest/include")
target_link_libraries(batch_dlr PRIVATE power_two "${XCA_BASELINE_BUILD}/lib/libgtest_main.a" "${XCA_BASELINE_BUILD}/lib/libgtest.a")
add_test(NAME batch_dlr COMMAND batch_dlr)
add_executable(batch_api test_api.cpp)
target_include_directories(batch_api PRIVATE "${XCA_BASELINE_BUILD}/deps/GTest_src/googletest/include")
target_link_libraries(batch_api PRIVATE power_two "${XCA_BASELINE_BUILD}/c++/triqs_xca/libtriqs_xca_c.a"
 "${XCA_BASELINE_BUILD}/lib/libgtest_main.a" "${XCA_BASELINE_BUILD}/lib/libgtest.a")
add_test(NAME batch_api COMMAND batch_api)
add_executable(fused_primitives test_fusion.cpp ../index_batch/device_ops.cpp)
target_include_directories(fused_primitives PRIVATE "${XCA_BASELINE_BUILD}/deps/GTest_src/googletest/include")
target_link_libraries(fused_primitives PRIVATE power_two "${XCA_BASELINE_BUILD}/lib/libgtest_main.a" "${XCA_BASELINE_BUILD}/lib/libgtest.a")
add_test(NAME fused_primitives COMMAND fused_primitives)
add_executable(primitive_benchmark primitive_benchmark.cpp ../index_batch/device_ops.cpp)
target_link_libraries(primitive_benchmark PRIVATE power_two)
