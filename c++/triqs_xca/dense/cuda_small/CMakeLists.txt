set(XCA_CUDA_SMALL AUTO CACHE STRING "Build small CUDA kernels: AUTO, ON or OFF")
set_property(CACHE XCA_CUDA_SMALL PROPERTY STRINGS AUTO ON OFF)
if(NOT XCA_CUDA_SMALL MATCHES "^(AUTO|ON|OFF)$")
  message(FATAL_ERROR "XCA_CUDA_SMALL must be AUTO, ON or OFF")
endif()
set(XCA_WITH_CUDA_SMALL OFF PARENT_SCOPE)
if(XCA_CUDA_SMALL STREQUAL "OFF")
  return()
endif()
if(NOT nda_WITH_CUDA_SUPPORT)
  if(XCA_CUDA_SMALL STREQUAL "ON")
    message(FATAL_ERROR "XCA_CUDA_SMALL=ON requires CUDA-enabled NDA/TRIQS")
  endif()
  message(STATUS "CUDA-small disabled: NDA has no CUDA support")
  return()
endif()
include(CheckLanguage)
check_language(CUDA)
if(NOT CMAKE_CUDA_COMPILER)
  if(XCA_CUDA_SMALL STREQUAL "ON")
    message(FATAL_ERROR "XCA_CUDA_SMALL=ON requires a CUDA compiler")
  endif()
  message(STATUS "CUDA-small disabled: no CUDA compiler; library CUDA remains available")
  return()
endif()
enable_language(CUDA)
find_package(CUDAToolkit REQUIRED)
add_library(triqs_xca_small_kernels STATIC kernels.cu)
set_target_properties(triqs_xca_small_kernels PROPERTIES CUDA_STANDARD 17
  CUDA_STANDARD_REQUIRED ON POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(triqs_xca_small_kernels PRIVATE XCA_POW2_DOT_TILE=4)
target_compile_options(triqs_xca_small_kernels PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=true>)
target_link_libraries(triqs_xca_small_kernels PUBLIC CUDA::cudart)
target_link_libraries(triqs_xca_c PUBLIC triqs_xca_small_kernels)
target_compile_definitions(triqs_xca_c PUBLIC XCA_HAVE_CUDA_SMALL)
install(TARGETS triqs_xca_small_kernels EXPORT triqs_xca-targets DESTINATION ${CMAKE_INSTALL_LIBDIR})
set(XCA_WITH_CUDA_SMALL ON PARENT_SCOPE)
message(STATUS "CUDA-small enabled; architectures=${CMAKE_CUDA_ARCHITECTURES}")
