set(XCA_CUDA_BLOCK_SPARSE AUTO CACHE STRING "Build ordered sparse CUDA kernels: AUTO, ON or OFF")
set_property(CACHE XCA_CUDA_BLOCK_SPARSE PROPERTY STRINGS AUTO ON OFF)
if(NOT XCA_CUDA_BLOCK_SPARSE MATCHES "^(AUTO|ON|OFF)$")
  message(FATAL_ERROR "XCA_CUDA_BLOCK_SPARSE must be AUTO, ON or OFF")
endif()
if(XCA_CUDA_BLOCK_SPARSE STREQUAL "OFF")
  return()
endif()
if(NOT nda_WITH_CUDA_SUPPORT)
  if(XCA_CUDA_BLOCK_SPARSE STREQUAL "ON")
    message(FATAL_ERROR "Ordered sparse kernels require CUDA-enabled NDA")
  endif()
  return()
endif()
include(CheckLanguage)
check_language(CUDA)
if(NOT CMAKE_CUDA_COMPILER)
  if(XCA_CUDA_BLOCK_SPARSE STREQUAL "ON")
    message(FATAL_ERROR "Ordered sparse kernels require a CUDA compiler")
  endif()
  return()
endif()
enable_language(CUDA)
find_package(CUDAToolkit REQUIRED)
add_library(triqs_xca_sparse_kernels STATIC kernels.cu)
set_target_properties(triqs_xca_sparse_kernels PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON POSITION_INDEPENDENT_CODE ON)
target_compile_options(triqs_xca_sparse_kernels PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
target_link_libraries(triqs_xca_sparse_kernels PUBLIC CUDA::cudart)
target_link_libraries(triqs_xca_c PUBLIC triqs_xca_sparse_kernels)
target_compile_definitions(triqs_xca_c PUBLIC XCA_HAVE_BLOCK_SPARSE_KERNELS)
install(TARGETS triqs_xca_sparse_kernels EXPORT triqs_xca-targets DESTINATION ${CMAKE_INSTALL_LIBDIR})
